4.3. Example 1: Time Complexity#

What is the time complexity of access, search, insertion, deletion of array data structure? Explain why.

import numpy as np

class MyArray:
    def __init__(self, input_list):
        self.array = np.array(input_list, dtype="i")

    def myInsert(self, x, i):
        new_array = np.empty(len(self.array) + 1, dtype="i")
        new_array[:len(self.array)] = self.array
        n = len(new_array)

        # Shift elements to the right
        for idx in range(n - 2, i - 1, -1):
            new_array[idx + 1] = new_array[idx]

        new_array[i] = x
        self.array = new_array  # Update self.array
        return self.array

    def myIndex(self, x):
        n = len(self.array)
        for idx in range(n):
            if self.array[idx] == x:
                return idx
        return -1  # If the key is not found

    def myDelete(self, i):
        n = len(self.array)

        # Shift elements to the left
        for idx in range(i + 1, n):
            self.array[idx - 1] = self.array[idx]

        self.array = self.array[:n - 1]  # Update self.array
        return self.array

    def myRemove(self, x):
        i = self.myIndex(x)
        if i == -1:
            print("Element not found.")
            return self.array

        print("Element found at position:", i)
        self.myDelete(i)
        return self.array

4.3.1. Solution#

  • access: \(O(1)\) - access directly by index

  • access: \(O(n)\) - search orderly

  • insertion: \(O(n)\) - need to move elements

  • deletion: \(O(n)\) - need to move elements

4.5. Example 3#

Create a function that output unique elements in a given array. Unique means distinct elements. What is the time complexity of the algorithm?

def unique(arr):
    result = []
    for i in arr:
        if i not in result:
            result.append(i)
    return result

a = [1, 2, 2, 3, 4, 5, 5, 6, 7, 7]
unique(a)

[1, 2, 3, 4, 5, 6, 7]

time complexity: O(n²)

4.6. Example 4#

Create a function that returns how many number of elements are odd and even elements in the array.

def EvenOdd(arr):
    counterodd=0
    countereven=0
    for i in arr:
        if i % 2 == 1:
            counterodd+=1
        elif i % 2 == 0:
            countereven+=1
    return counterodd, countereven

a = [1, 2, 2, 3, 4, 5, 5, 6, 7, 7]
EvenOdd(a)

4.7. A new question#

Create a function find_mode(arr) that takes an array of integers as input and returns the mode of the array. The mode is the value that appears most frequently in the array. If there are multiple values that appear with the same highest frequency, return the smallest one.

4.7.1. solution#

def find_mode(arr):
    if not arr:
        return None

    # Sort the array
    arr.sort()

    current_num = arr[0]
    current_count = 1
    max_count = 1
    mode = current_num

    # Iterate through the sorted array
    for i in range(1, len(arr)):
        if arr[i] == current_num:
            current_count += 1
        else:
            # If we found a new element
            if current_count > max_count:
                max_count = current_count
                mode = current_num

            current_num = arr[i]
            current_count = 1

    # Check the last group
    if current_count > max_count:
        mode = current_num

    return mode

arr=[5, 5, 2, 2, 1, 1]
find_mode(arr)
